\section{Introduction}

\begin{frame}{Introduction: Diabetic Retinopathy (DR)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\small
        \item DR is a serious diabetes complication that damages retinal vessels
        \item Progresses without symptoms --- patients often notice late
        \item Leading cause of \alert{preventable blindness} in working-age adults
      \end{itemize}
      \vspace{4pt}
      \begin{block}{Global Burden}
        \small
        \begin{itemize}
          \item \textbf{103 million} people affected today
          \item Projected to reach \textbf{700 million by 2045}
          \item India is among the \textbf{top 3 nations} for diabetes burden
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Five DR Severity Stages}
        \small\renewcommand{\arraystretch}{1.2}
        \begin{tabular}{cl}
          \textbf{Grade 0} & No DR \\
          \textbf{Grade 1} & Mild NPDR \\
          \textbf{Grade 2} & Moderate NPDR \\
          \textbf{Grade 3} & Severe NPDR \\
          \textbf{Grade 4} & Proliferative DR \\
        \end{tabular}
      \end{block}
      \begin{alertblock}{Key Challenge}\small
        Early stages are asymptomatic --- automated screening at scale is urgently needed
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 4 — Why FL?
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Why Federated Learning for DR?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Problem with Centralized AI}
        \small\begin{itemize}
          \item Deep learning needs large, diverse datasets
          \item Patient retinal images cannot be freely shared
          \item HIPAA (USA) and GDPR (EU) restrict cross-institutional data sharing
          \item Single central data store = single privacy risk
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Federated Learning Solution}
        \small\begin{itemize}
          \item Each hospital keeps patient data locally
          \item Only \textbf{model updates} (not images) are shared
          \item Global model is trained collaboratively
          \item Multi-hospital collaboration while staying compliant
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \begin{block}{Remaining Challenges in FL for DR}\small
    Standard FedAvg suffers from \textbf{client drift} on Non-IID hospital data $\Rightarrow$ slow, unstable convergence. Basic FL has no \textbf{formal privacy guarantees}. Most systems are not tested on the \textbf{Indian patient population}.
  \end{block}
\end{frame}

\section{Problem Statement}
% ═══════════════════════════════════════════════════════════════════════════════

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 5 — Problem Statement
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Problem Statement}
  \begin{alertblock}{Core Problem}\small
    Standard FL-based DR systems use FedAvg on assumed IID data, without formal privacy guarantees, and have not been tested on Indian patient populations using the IRDRD dataset.
  \end{alertblock}
  \vspace{6pt}
  \textbf{Specific Issues:}
  \begin{enumerate}\small
    \item FedAvg fails on \textbf{Non-IID} hospital data $\rightarrow$ client drift $\rightarrow$ poor convergence
    \item No FL paper has applied \textbf{SCAFFOLD} to DR detection
    \item Most FL systems have \textbf{no formal privacy} (DP) beyond gradient sharing
    \item No federated study uses the \textbf{IRDRD dataset} (Indian population)
    \item \textbf{Personalization} is rarely combined with privacy-preserving aggregation
    \item No paper gives an empirical \textbf{FedAvg vs.\ SCAFFOLD} comparison on retinal data
  \end{enumerate}
\end{frame}

    \item[\textbf{O4.}] Add \textbf{Personalization} via local fine-tuning to adapt the global model to each hospital
    \item[\textbf{O5.}] Conduct a direct \textbf{FedAvg vs.\ SCAFFOLD} comparison under identical Non-IID conditions
    \item[\textbf{O6.}] Deliver a privacy-compliant system satisfying HIPAA/GDPR requirements
  \end{enumerate}
\end{frame}

\section{Proposed Methodology}
% ═══════════════════════════════════════════════════════════════════════════════

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 11 — System Overview
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Proposed System: Architecture Overview}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Central Server}\small
        \begin{itemize}
          \item Hosts global CNN model
          \item Maintains SCAFFOLD control variates $c$
          \item Aggregates $\Delta w_k$ and $\Delta c_k$ from clients
          \item Broadcasts updated $w^{t+1}$ and $c^{t+1}$
        \end{itemize}
      \end{block}
      \vspace{4pt}
      \begin{block}{Each Client (Hospital Node)}\small
        \begin{itemize}
          \item Holds local IRDRD partition (Non-IID)
          \item Trains local CNN with DP-SGD
          \item Applies SCAFFOLD gradient correction
          \item Sends only weight updates (no images)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{Three-Layer Privacy}\small
        \begin{enumerate}
          \item \textbf{FL Level:} No raw images shared
          \item \textbf{DP Level:} Gaussian noise on gradients ($\epsilon \leq 5$)
          \item \textbf{Personalization:} DP applied during local fine-tuning too
        \end{enumerate}
      \end{alertblock}
      \vspace{4pt}
      \begin{exampleblock}{System Stack}\small
        PyTorch + Flower (flwr)\\
        Opacus (DP-SGD)\\
        OpenCV + Albumentations\\
        5 clients, 100 rounds\\
        IRDRD dataset (Non-IID, $\alpha=0.5$)
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 12 — Workflow
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Step-by-Step Federated Workflow}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{enumerate}\small
        \item[\textbf{1.}] \textbf{Dataset Preparation} --- IRDRD split into 5 Non-IID subsets via Dirichlet sampling ($\alpha = 0.5$)
        \item[\textbf{2.}] \textbf{Preprocessing (local)} --- Resize $224\times224$, CLAHE, normalize, augment
        \item[\textbf{3.}] \textbf{Model Init} --- Server initializes CNN + SCAFFOLD control variates $c^0 = 0$
        \item[\textbf{4.}] \textbf{Local Training (DP-SGD)} --- Gradient clipping + Gaussian noise; SCAFFOLD corrects gradient direction
      \end{enumerate}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{enumerate}\small
        \item[\textbf{5.}] \textbf{Transmit Updates} --- Send $\Delta w_k$ and $\Delta c_k$ only (no images)
        \item[\textbf{6.}] \textbf{Server Aggregation} --- Aggregate weight and control variate updates
        \item[\textbf{7.}] \textbf{Broadcast} --- Updated $w^{t+1}$, $c^{t+1}$ sent to all clients
        \item[\textbf{8.}] Repeat for \textbf{100 rounds} or until convergence
        \item[\textbf{9.}] \textbf{Personalization} --- Freeze feature extractor; fine-tune FC layers locally (10 epochs, DP-SGD)
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 14 — SCAFFOLD Algorithm
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{SCAFFOLD Algorithm: How It Works}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{alertblock}{Problem: Client Drift in FedAvg}\small
        On Non-IID data, each client's local gradient pulls the model toward its own local optimum, not the global one. After $E$ local epochs, models diverge significantly.
      \end{alertblock}
      \vspace{4pt}
      \begin{block}{FedAvg Aggregation}\small
        \[
        w^{t+1} = \sum_{k=1}^{N} \frac{n_k}{n}\, w_k^{t, \text{local}}
        \]
        Simple weighted average --- does not correct drift.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{exampleblock}{SCAFFOLD: Corrected Gradient Step}\small
        Each local SGD step uses a corrected gradient:
        \[
        w_k \leftarrow w_k - \eta_l (g_k - c_k + c)
        \]
        $c_k$ = local control variate, $c$ = global control variate.
      \end{exampleblock}
      \vspace{4pt}
      \begin{exampleblock}{SCAFFOLD Server Update}\small
        \[
        w^{t+1} = w^t + \frac{\eta_s}{N}\sum_k \Delta w_k
        \]
        \[
        c^{t+1} = c^t + \frac{1}{N}\sum_k \Delta c_k
        \]
        Both weights and control variates are updated.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \begin{center}
    \small \textbf{SCAFFOLD complexity:} $\mathcal{O}(1/\epsilon^2)$ vs.\ $\mathcal{O}(1/\epsilon^{2.5})$ for FedAvg on Non-IID data
  \end{center}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
% SLIDE 15 — Differential Privacy + Personalization
% ─────────────────────────────────────────────────────────────────────────────
\begin{frame}{Differential Privacy and Personalization}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Differential Privacy via DP-SGD}\small
        Applied during local training at each client:
        \begin{enumerate}
          \item Compute per-sample gradients $g_i(w)$
          \item \textbf{Clip:} $\tilde{g}_i = g_i \big/ \max(1, \|g_i\|_2 / C)$
          \item \textbf{Add noise:} $\hat{g} = \frac{1}{B}\!\left(\sum_i \tilde{g}_i + \mathcal{N}(0, \sigma^2 C^2 \mathbf{I})\right)$
          \item Update weights with noisy gradient
        \end{enumerate}
        \vspace{2pt}
        $C=1.0$, $\sigma=1.1$, $\delta=10^{-5}$, $\epsilon \leq 5$
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Personalization via Local Fine-Tuning}\small
        After global convergence:
        \begin{itemize}
          \item \textbf{Freeze:} Conv Blocks 1--4 + GAP
          \item \textbf{Fine-tune:} FC1, FC2, Output layers
          \item \textbf{Data:} Local IRDRD partition only
          \item \textbf{Duration:} 10 local epochs
          \item \textbf{LR:} Reduced to $\eta = 1\times10^{-5}$
          \item DP-SGD applied here too
        \end{itemize}
        \vspace{2pt}
        Result: Global features + local calibration
      \end{block}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \begin{center}
    \small
    \textbf{Privacy guarantee:} No raw images shared at any stage. DP-SGD ensures $(\epsilon, \delta)$-DP during both federated training and personalization.
  \end{center}
\end{frame}

        \item Five-class ICDR severity labels (Grade 0--4)

        \item Weighted random sampling for class imbalance
        \item \textbf{Zero images transmitted to server}

\begin{frame}{Conclusion}
  \begin{itemize}\small
    \item DR is a major global health challenge requiring scalable, privacy-preserving automated detection
    \item Standard FL (FedAvg) is insufficient for real-world hospital settings due to Non-IID client drift
    \item The proposed system \textbf{combines SCAFFOLD, personalization, and differential privacy} in a single federated CNN framework for 5-class DR grading
    \item Evaluated on the \textbf{IRDRD dataset} --- the first FL-based DR study to do so
    \item A controlled \textbf{FedAvg vs.\ SCAFFOLD comparison} will provide the first empirical evidence of SCAFFOLD's advantage in medical imaging
    \item Expected: \textbf{88--94\% accuracy} with SCAFFOLD + personalization; formal $(\epsilon \leq 5, \delta=10^{-5})$-DP guarantee; privacy-compliant deployment
  \end{itemize}
  \vspace{4pt}
  \begin{block}{Future Work}\small
    Integrate Grad-CAM for model explainability; deploy in a real multi-hospital network; explore adaptive noise scheduling for a better accuracy-privacy tradeoff
  \end{block}
\end{frame}
